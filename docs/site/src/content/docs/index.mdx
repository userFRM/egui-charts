<script is:inline set:html={`
  (() => {
    const video = document.getElementById('tour-video');
    const play = document.getElementById('tour-play');
    const motion = window.matchMedia('(prefers-reduced-motion: reduce)');
    const load = () => {
      for (const source of video.querySelectorAll('source')) source.src = source.dataset.src;
      video.load();
    };
    const update = () => {
      video.autoplay = !motion.matches;
      play.hidden = !motion.matches;
      video.controls = !motion.matches;
      if (motion.matches) {
        video.pause();
        for (const source of video.querySelectorAll('source')) source.removeAttribute('src');
        video.load();
      } else {
        load();
        video.play().catch(() => { play.hidden = false; });
      }
    };
    play.addEventListener('click', () => {
      video.controls = true;
      load();
      video.play().then(() => { play.hidden = true; }).catch(() => { play.hidden = false; });
    });
    motion.addEventListener('change', update);
    update();
  })();
`} />
